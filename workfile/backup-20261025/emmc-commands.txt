# merlinx eMMC / MSDC data collection
# Run these on the Mac/PC where adb sees the phone.
# The phone must allow root (su).

adb shell su -c 'mount -t debugfs none /sys/kernel/debug 2>/dev/null'

adb shell su -c 'dmesg | grep -iE "msdc|mmc0|mmc1|emmc|hs400|vemc|vmch|vmmc|vqmmc"' > merlinx-emmc-dmesg.txt
adb shell su -c 'cat /proc/partitions' > merlinx-emmc-partitions.txt
adb shell su -c 'ls -l /sys/class/mmc_host' > merlinx-emmc-hosts.txt
adb shell su -c 'cat /sys/kernel/debug/mmc0/ios 2>/dev/null' > merlinx-emmc-ios.txt
adb shell su -c 'cat /sys/kernel/debug/regulator/regulator_summary 2>/dev/null' > merlinx-emmc-regulators.txt
adb shell su -c 'cat /sys/kernel/debug/pinctrl/*/pinmux-pins 2>/dev/null' > merlinx-emmc-pinmux.txt
adb shell su -c 'cat /proc/cmdline' > merlinx-emmc-cmdline.txt
adb shell getprop ro.boot.boot_devices > merlinx-emmc-bootdevices.txt

# Optional: one combined file for sending
cat merlinx-emmc-*.txt > merlinx-emmc-all.txt
